<Project>
  <Title>Ideas Hub – Intelligent Project Showcase Platform</Title>

  <Tagline>“The smarter way to showcase your projects — connect GitHub, auto-generate your portfolio, and let AI summarize your work.”</Tagline>

  <Summary>
    Ideas Hub is a web-based platform designed to help students, developers, and professionals easily showcase their projects online. 
    Users can either add their projects manually or log in using GitHub to automatically import their repositories and create interactive showcase pages. 
    The platform optionally integrates AI to generate intelligent summaries and auto-tags, making every portfolio smart and professional.
  </Summary>

  <ConceptAndPurpose>
    <Goal>
      <Developers>Empower developers to create professional project showcases without complex setup.</Developers>
      <Automation>Automatically extract project details from GitHub repositories using APIs.</Automation>
      <Innovation>Integrate AI summarization to generate concise project overviews and tags.</Innovation>
      <Sharing>Generate unique, shareable project links for professional visibility.</Sharing>
    </Goal>
  </ConceptAndPurpose>

  <BeyondAIConcept>
    <Definition>Beyond AI = Automation + Smart Summarization + Real Integration.</Definition>
    <Highlights>
      <Point>Uses GitHub API to fetch real repository data.</Point>
      <Point>Uses AI to auto-summarize README files and generate tags.</Point>
      <Point>Creates shareable showcase pages with unique URLs.</Point>
      <Point>Combines human (manual add) + machine (auto-import) workflow.</Point>
    </Highlights>
  </BeyondAIConcept>

  <CoreFeatures>
    <Feature type="ManualAdd">Users can manually add project details via form.</Feature>
    <Feature type="GitHubLogin">Log in with GitHub using NextAuth OAuth.</Feature>
    <Feature type="ImportFromGitHub">Import repository details like title, description, tech stack.</Feature>
    <Feature type="AISummary">Auto-generate project summaries using AI from README.md.</Feature>
    <Feature type="AutoTagging">Generate tags automatically from detected tech stack.</Feature>
    <Feature type="MyProjects">Display all imported and manually added projects.</Feature>
    <Feature type="ShareableShowcase">Create unique shareable URLs for each project.</Feature>
    <Feature type="ExploreSearch">Search and filter projects by name, tech, or tags.</Feature>
    <Feature type="ResponsiveUI">Fully responsive modern UI using Tailwind CSS.</Feature>
    <Feature type="UserProfile">Users have bio, profile picture, and social links.</Feature>
  </CoreFeatures>

  <TechStack>
    <Frontend>Next.js + Tailwind CSS</Frontend>
    <Authentication>NextAuth.js (GitHub OAuth)</Authentication>
    <Database>Supabase (Auth + DB + File Storage)</Database>
    <Integration>GitHub REST API</Integration>
    <AI>OpenAI API for project summaries and tag generation</AI>
    <Hosting>Vercel</Hosting>
  </TechStack>
